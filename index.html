local character = script.Parent
local humanoid = character:WaitForChild("Humanoid")

humanoid.CameraOffset = Vector3.new(0, 0.2, -0.8)

2. Корсар 1: Анимация и зажигалка

Куда: Внутрь Tool (LocalScript)
local tool = script.Parent
local replicatedStorage = game:GetService("ReplicatedStorage")
local petardEvent = replicatedStorage:WaitForChild("PetardEvent")

local lightAnim = Instance.new("Animation")
lightAnim.AnimationId = "rbxassetid://ВАШ_ID_АНИМАЦИИ"

local state = 0

tool.Activated:Connect(function()
	local character = tool.Parent
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")
	if not humanoid then return end

	if state == 0 then
		state = 1
		petardEvent:FireServer("Light")

		local animator = humanoid:FindFirstChildOfClass("Animator")
		if animator then
			local track = animator:LoadAnimation(lightAnim)
			track.Priority = Enum.AnimationPriority.Action
			track:Play()
		end

		local leftHand = character:FindFirstChild("LeftHand") or character:FindFirstChild("Left Arm")
		local lighterTemplate = replicatedStorage:FindFirstChild("Lighter")
		if leftHand and lighterTemplate then
			local lighter = lighterTemplate:Clone()
			lighter.CanCollide = false
			lighter.Massless = true
			lighter.CFrame = leftHand.CFrame
			lighter.Parent = character

			local weld = Instance.new("WeldConstraint")
			weld.Part0 = leftHand
			weld.Part1 = lighter
			weld.Parent = lighter

			task.delay(1, function()
				if lighter then lighter:Destroy() end
			end)
		end

	elseif state == 1 then
		state = 2
		petardEvent:FireServer("Throw")
	end
end)

3. Корсар 1: Взрыв, звук и урон

Куда: Внутрь Tool (обычный Script)
local tool = script.Parent
local handle = tool:WaitForChild("Handle")
local replicatedStorage = game:GetService("ReplicatedStorage")
local debris = game:GetService("Debris")
local petardEvent = replicatedStorage:WaitForChild("PetardEvent")

local fuseTime = 4
local blastRadius = 18
local damageRadius = 5
local maxDamage = 10
local throwForce = 55

local isLit = false
local isThrown = false

local function detonate()
	if not tool or not tool.Parent then return end
	local explosionPos = handle.Position

	local soundPart = Instance.new("Part")
	soundPart.Transparency = 1
	soundPart.Anchored = true
	soundPart.CanCollide = false
	soundPart.Position = explosionPos
	soundPart.Parent = workspace

	local expSoundTemplate = handle:FindFirstChild("ExplosionSound")
	if expSoundTemplate then
		local sound = expSoundTemplate:Clone()
		sound.Parent = soundPart
		sound:Play()
	end
	debris:AddItem(soundPart, 3)

	local explosion = Instance.new("Explosion")
	explosion.Position = explosionPos
	explosion.BlastRadius = blastRadius
	explosion.BlastPressure = 2000
	explosion.DestroyJointRadiusPercent = 0
	explosion.Parent = workspace

	for _, player in ipairs(game:GetService("Players"):GetPlayers()) do
		local char = player.Character
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		local root = char and char:FindFirstChild("HumanoidRootPart")

		if hum and root then
			local distance = (root.Position - explosionPos).Magnitude
			if distance <= damageRadius then
				local damage = math.ceil(maxDamage * (1 - (distance / damageRadius)))
				hum:TakeDamage(damage)
			end
		end
	end

	tool:Destroy()
end

petardEvent.OnServerEvent:Connect(function(player, action)
	if tool.Parent ~= player.Character then return end

	if action == "Light" and not isLit then
		isLit = true
		local sparks = handle:FindFirstChild("Sparks")
		if sparks then sparks.Enabled = true end

		local fuseSound = handle:FindFirstChild("FuseSound")
		if fuseSound then fuseSound:Play() end

		task.delay(fuseTime, detonate)

	elseif action == "Throw" and isLit and not isThrown then
		isThrown = true
		local rootPart = player.Character:FindFirstChild("HumanoidRootPart")

		tool.Parent = workspace
		handle.CanCollide = true

		if rootPart then
			local lookVector = rootPart.CFrame.LookVector
			handle.AssemblyLinearVelocity = (lookVector * throwForce) + Vector3.new(0, 15, 0)
		end
	end
end)

4. Автоматика поезда (Waypoints и плавное торможение)

Куда: Внутрь модели поезда (обычный Script)
local RunService = game:GetService("RunService")

local train = script.Parent
local root = train.PrimaryPart
local waypointsFolder = workspace:WaitForChild("Waypoints")

local MAX_SPEED = 45
local ACCELERATION = 10
local DECELERATION = 12
local BRAKE_DISTANCE = 60

local currentSpeed = 0
local currentIndex = 1

local waypoints = {}
for _, point in ipairs(waypointsFolder:GetChildren()) do
	local num = tonumber(point.Name)
	if num then
		waypoints[num] = point
	end
end

if waypoints[1] then
	train:PivotTo(waypoints[1].CFrame)
end

local function getDistanceToNextStation(startIndex, currentPos)
	local totalDist = 0
	local prevPos = currentPos

	for i = startIndex, #waypoints do
		local wp = waypoints[i]
		totalDist += (wp.Position - prevPos).Magnitude
		prevPos = wp.Position

		local waitObj = wp:FindFirstChild("WaitTime")
		if waitObj and waitObj.Value > 0 then
			return totalDist
		end
	end

	return math.huge
end

task.spawn(function()
	while true do
		local nextIndex = currentIndex + 1
		if nextIndex > #waypoints then
			nextIndex = 1
		end

		local targetPoint = waypoints[nextIndex]
		if not targetPoint then break end

		while true do
			local dt = RunService.Heartbeat:Wait()
			local currentPos = root.Position
			local targetPos = targetPoint.Position
			local vectorToTarget = targetPos - currentPos
			local distToPoint = vectorToTarget.Magnitude

			local distToStation = getDistanceToNextStation(nextIndex, currentPos)
			local targetSpeed = MAX_SPEED

			if distToStation <= BRAKE_DISTANCE then
				targetSpeed = math.max(4, MAX_SPEED * (distToStation / BRAKE_DISTANCE))
			end

			if currentSpeed < targetSpeed then
				currentSpeed = math.min(targetSpeed, currentSpeed + ACCELERATION * dt)
			elseif currentSpeed > targetSpeed then
				currentSpeed = math.max(targetSpeed, currentSpeed - DECELERATION * dt)
			end

			local step = currentSpeed * dt

			if distToPoint <= step then
				train:PivotTo(CFrame.lookAt(targetPos, targetPos + vectorToTarget.Unit))
				currentIndex = nextIndex
				break
			else
				local direction = vectorToTarget.Unit
				local newPos = currentPos + direction * step
				local targetCFrame = CFrame.lookAt(newPos, newPos + direction)
				train:PivotTo(root.CFrame:Lerp(targetCFrame, 0.15))
			end
		end

		local waitObj = targetPoint:FindFirstChild("WaitTime")
		if waitObj and waitObj.Value > 0 then
			currentSpeed = 0
			task.wait(waitObj.Value)
		end
	end
end)
